# Publisher: two STOCK attributes and two CUSTOM ones the VR-Link RPR mapper does not model.
# Custom attributes are the entire point of REQ-08 - the C# agent could not put these on the wire.
JOIN exec=BTK-NativeTest fed=BTK-CustomRPR1.fed federate=NativePub create=1
CLASS name=BaseEntity attrs=EntityIdentifier,WorldLocation,BtkCustomValue,BtkCustomTag mode=pub
REGISTER id=t1 class=BaseEntity name=BTK-TANK-01
WAIT ms=1500
SETATTR id=t1 name=EntityIdentifier hex=000100010065
SETATTR id=t1 name=WorldLocation hex=c14491b80000000040934a00000000004154fb1800000000
SETATTR id=t1 name=BtkCustomValue f64=1234.5
SETATTR id=t1 name=BtkCustomTag str=CUSTOM-TAG-01
UPDATE id=t1
WAIT ms=4000
QUIT
